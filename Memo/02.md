#pragma once // 使头文件仅编译一次

#include "CoreMinimal.h" // 引入虚幻基础工具
#include "Materials/MaterialExpression.h"  // 引入材质节点的最基础父类
#include "MaterialExpressionTranslucentSceneColor.generated.h" // unreal生成文件

// 上述为头文件基本的固定配置

// 注册节点
UCLASS(
    CollapseCategories,
    HideCategories=Object,
    meta=(DisplayName="Translucent SceneColor")
)

// 定义为一个C++类 命名
class UMaterialExpressionTranslucentSceneColor
    : public UMaterialExpression // 继承于基础材质节点父类
{
    GENERATED_BODY() // 类的内容由此开始

public: // public以下的内容可以被外部访问 因为需要编译器去调用

    // 仅声明构造函数 具体函数逻辑在 .cpp 中执行
    UMaterialExpressionTranslucentSceneColor(
        const FObjectInitializer& ObjectInitializer
    ); 

    // UPROPERTY 告诉引擎以下的变量需要显示和保存在编辑器中
    UPROPERTY(
        meta=(
            RequiredInput="true",
            ToolTip="Vector2 UV Coordinates"
        )
    )
    FExpressionInput UV; // 创建一个名为UV的输入引脚

#if WITH_EDITOR

    // 传统材质编译入口
    virtual int32 Compile(
        FMaterialCompiler* Compiler,
        int32 OutputIndex
    ) override;

    // UE 5.8 Material IR 编译入口
    virtual void Build(
        MIR::FEmitter& Emitter
    ) override;

    // 节点方框上的名称
    virtual void GetCaption(
        TArray<FString>& OutCaptions
    ) const override;

    // 输入引脚名称
    virtual FName GetInputName(
        int32 InputIndex
    ) const override;

    // 指定 UV 输入是 Vector2
    virtual EMaterialValueType GetInputValueType(
        int32 InputIndex
    ) override;

    // 指定 RGB 输出是 Vector3
    virtual EMaterialValueType GetOutputValueType(
        int32 OutputIndex
    ) override;

#endif
};
